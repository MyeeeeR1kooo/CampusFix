24def _shanghai_date(value: datetime) -> date:

30def get_analytics(db: Session, *, now: datetime | None = None) -> AnalyticsResponse: 

31 """Aggregate current ticket data directly from PostgreSQL.""" 

32 status_rows = db.execute( 

33 select(Ticket.status, func.count(Ticket.id)).group_by(Ticket.status) 

34 ).all() 

35 status_counts = {status: int(count) for status, count in status_rows} 

36 

37 category_rows = db.execute( 

38 select(Ticket.category, func.count(Ticket.id)).group_by(Ticket.category) 

39 ).all() 

40 category_counts = {category: int(count) for category, count in category_rows} 

41 

42 building_rows = db.execute( 

43 select(Location.building, func.count(Ticket.id)) 

44 .join(Ticket, Ticket.location_id == Location.id) 

45 .group_by(Location.building) 

46 .order_by(Location.building.asc()) 

47 ).all() 

48 

49 backlog = db.scalar( 

50 select(func.count(Ticket.id)).where( 

51 Ticket.status.not_in(("CLOSED", "REJECTED", "CANCELLED")) 

52 ) 

53 ) or 0 

54 average = db.scalar( 

55 select(func.avg(extract("epoch", Ticket.closed_at - Ticket.created_at))).where( 

56 Ticket.status == "CLOSED", Ticket.closed_at.is_not(None) 

57 ) 

58 ) 

59 

60 today = _shanghai_date(now or datetime.now(timezone.utc)) 

61 first_day = today - timedelta(days=29) 

62 # PostgreSQL's timezone() converts stored UTC timestamps before date casting. 

63 local_created = cast(func.timezone("Asia/Shanghai", Ticket.created_at), Date) 

64 local_closed = cast(func.timezone("Asia/Shanghai", Ticket.closed_at), Date) 

65 created_rows = db.execute( 

66 select(local_created, func.count(Ticket.id)) 

67 .where(local_created >= first_day, local_created <= today) 

68 .group_by(local_created) 

69 ).all() 

70 closed_rows = db.execute( 

71 select(local_closed, func.count(Ticket.id)) 

72 .where( 

73 Ticket.status == "CLOSED", Ticket.closed_at.is_not(None), 

74 local_closed >= first_day, local_closed <= today, 

75 ) 

76 .group_by(local_closed) 

77 ).all() 

78 created_by_day = {day: int(count) for day, count in created_rows} 

79 closed_by_day = {day: int(count) for day, count in closed_rows} 

80 

81 return AnalyticsResponse( 

82 by_status=[StatusCount(status=status, count=status_counts.get(status, 0)) for status in STATUSES], 

83 by_category=[CategoryCount(category=category, count=category_counts.get(category, 0)) for category in CATEGORIES], 

84 by_building=[BuildingCount(building=building, count=int(count)) for building, count in building_rows], 

85 backlog_count=int(backlog), 

86 average_close_seconds=float(average or 0), 

87 daily_trend=[ 

88 DailyTrend( 

89 date=day, 

90 created_count=created_by_day.get(day, 0), 

91 closed_count=closed_by_day.get(day, 0), 

92 ) 

93 for day in (first_day + timedelta(days=offset) for offset in range(30)) 

94 ], 

95 )
